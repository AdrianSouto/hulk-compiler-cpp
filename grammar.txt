BooleanTermTail -> ε

AdditiveExpressionTail -> MINUS MultiplicativeExpression AdditiveExpressionTail

FunctionBody -> ε

ParameterList -> IDENTIFIER TypeAnnotationOpt ParameterListTail
ParameterList -> ε

ParameterListTail -> ε

TypeAnnotationOpt -> ε

TypeName -> IDENTIFIER

DestructiveExpressionTail -> ε

AssignmentList -> AssignmentDecl AssignmentListTail

ElifList -> ε

TypeInstantiation -> NEW IDENTIFIER FunctionCallTail

ArgTypeOpt -> ε

MemberList -> ε

MemberListTail -> ε
