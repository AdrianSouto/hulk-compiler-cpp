# HULK Language Grammar - LL(1) Compatible with elif support + types with parameters
# Based on working version, adding type-related functionality with type parameters

Terminals: FUNCTION, TYPE, INHERITS, NEW, BASE, IF, ELIF, ELSE, WHILE, FOR, IN, IS, AS, LET, PRINT, TRUE, FALSE, TYPE_NUMBER, TYPE_STRING, TYPE_BOOLEAN, IDENTIFIER, NUMBER, STRING, LPAREN, RPAREN, LBRACE, RBRACE, SEMICOLON, COMMA, COLON, DOT, PLUS, MINUS, MULTIPLY, DIVIDE, MODULO, POWER, CONCAT, CONCAT_SPACE, EQUALS, ASSIGN, EQEQ, NOTEQ, LESS, LESSEQ, GREATER, GREATEREQ, AND, OR, NOT, ARROW, EOF

NonTerminals: Program, StmtList, Stmt, Expr, LogicalOr, LogicalOrRest, LogicalAnd, LogicalAndRest, Equality, EqualityRest, Relational, RelationalRest, Additive, AdditiveRest, Multiplicative, MultiplicativeRest, Power, PowerRest, Unary, Primary, FuncDef, ParamList, ParamRest, Block, ExprList, ArgList, ArgRest, VarDecl, IfExpr, ElifList, WhileLoop, ForLoop, TypeInst, AccessChain, FuncBody, AssignmentList, AssignmentRest, AssignmentDecl, AssignmentTail, TypeDef, TypeAnnotationOpt, InheritsClause, MemberList, Member, MemberTail, TypeName, TypeParams

Productions:
Program -> StmtList EOF

StmtList -> Stmt StmtList | ε

Stmt -> Expr SEMICOLON | FuncDef SEMICOLON | TypeDef SEMICOLON

Expr -> VarDecl | IfExpr | WhileLoop | ForLoop | TypeInst | LogicalOr AssignmentTail

AssignmentTail -> ASSIGN Expr | ε

LogicalOr -> LogicalAnd LogicalOrRest

LogicalOrRest -> OR LogicalAnd LogicalOrRest | ε

LogicalAnd -> Equality LogicalAndRest

LogicalAndRest -> AND Equality LogicalAndRest | ε

Equality -> Relational EqualityRest

EqualityRest -> EQEQ Relational EqualityRest | NOTEQ Relational EqualityRest | ε

Relational -> Additive RelationalRest

RelationalRest -> LESS Additive RelationalRest | LESSEQ Additive RelationalRest | GREATER Additive RelationalRest | GREATEREQ Additive RelationalRest | IS TypeName | AS TypeName | ε

Additive -> Multiplicative AdditiveRest

AdditiveRest -> PLUS Multiplicative AdditiveRest | MINUS Multiplicative AdditiveRest | CONCAT Multiplicative AdditiveRest | CONCAT_SPACE Multiplicative AdditiveRest | ε

Multiplicative -> Power MultiplicativeRest

MultiplicativeRest -> MULTIPLY Power MultiplicativeRest | DIVIDE Power MultiplicativeRest | MODULO Power MultiplicativeRest | ε

Power -> Unary PowerRest

PowerRest -> POWER Unary PowerRest | ε

Unary -> NOT Unary | MINUS Unary | Primary

Primary -> LPAREN Expr RPAREN | NUMBER | STRING | TRUE | FALSE | IDENTIFIER AccessChain | PRINT LPAREN Expr RPAREN | Block | BASE AccessChain

AccessChain -> LPAREN ArgList RPAREN AccessChain | DOT IDENTIFIER AccessChain | ε

FuncDef -> FUNCTION IDENTIFIER LPAREN ParamList RPAREN TypeAnnotationOpt FuncBody

FuncBody -> ARROW Expr | Block

ParamList -> IDENTIFIER TypeAnnotationOpt ParamRest | ε

ParamRest -> COMMA IDENTIFIER TypeAnnotationOpt ParamRest | ε

Block -> LBRACE ExprList RBRACE

ExprList -> Expr SEMICOLON ExprList | ε

ArgList -> Expr ArgRest | ε

ArgRest -> COMMA Expr ArgRest | ε

VarDecl -> LET AssignmentList IN Expr

AssignmentList -> AssignmentDecl AssignmentRest

AssignmentRest -> COMMA AssignmentDecl AssignmentRest | ε

AssignmentDecl -> IDENTIFIER TypeAnnotationOpt EQUALS Expr

IfExpr -> IF LPAREN Expr RPAREN Expr ElifList ELSE Expr

ElifList -> ELIF LPAREN Expr RPAREN Expr ElifList | ε

WhileLoop -> WHILE LPAREN Expr RPAREN Expr

ForLoop -> FOR LPAREN IDENTIFIER IN Expr RPAREN Expr

TypeInst -> NEW IDENTIFIER LPAREN ArgList RPAREN

TypeDef -> TYPE IDENTIFIER TypeParams InheritsClause LBRACE MemberList RBRACE

TypeParams -> LPAREN ParamList RPAREN | ε

TypeAnnotationOpt -> COLON TypeName | ε

InheritsClause -> INHERITS IDENTIFIER | ε

MemberList -> Member MemberList | ε

Member -> IDENTIFIER TypeAnnotationOpt MemberTail SEMICOLON

MemberTail -> EQUALS Expr | LPAREN ParamList RPAREN TypeAnnotationOpt FuncBody | ε

TypeName -> IDENTIFIER | TYPE_NUMBER | TYPE_STRING | TYPE_BOOLEAN
